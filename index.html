function renderStationList() { const container = document.getElementById("stationList"); const keyword = document.getElementById("searchInput").value.trim().toLowerCase(); let renderData = [...radioDataSource]; // ===== 有搜索词时 ===== if (keyword) { // 只有在"广东"或"全部"分类下，才全局搜索 if (currentFilter === 'cat_gd' || currentFilter === 'cat_all') { renderData = radioDataSource.filter(item => item.name.toLowerCase().includes(keyword) || (item.city && item.city.toLowerCase().includes(keyword)) ); } else { // 其他分类：先全局搜索，再按当前分类筛选（保持分类内搜索） let allSearch = radioDataSource.filter(item => item.name.toLowerCase().includes(keyword) || (item.city && item.city.toLowerCase().includes(keyword)) ); const cat = categoryList.find(c => c.id === currentFilter); if (cat) { const name = cat.name; renderData = allSearch.filter(item => item.tag === name || item.area === name); } else { renderData = []; } } } else { // ===== 无搜索词：按分类显示 ===== if (currentFilter === "cat_favorites") { renderData = renderData.filter(item => favList.includes(item.id)); } else if (currentFilter === "cat_all") { // 全部显示 } else if (currentFilter === "cat_history") { renderData = renderData.filter(item => historyList.includes(item.id)); } else { const cat = categoryList.find(c => c.id === currentFilter); if (cat) { const name = cat.name; renderData = renderData.filter(item => item.tag === name || item.area === name); } else { renderData = []; } } } // ===== 后续渲染不变 ===== container.innerHTML = ""; if (renderData.length === 0) { container.innerHTML = `
📭 没有找到电台
`; return; } const isAllCategory = (currentFilter === 'cat_all'); renderData.forEach((radio) => { const isPlaying = radio.id === currentPlayingId && !(cardError && cardError.id === radio.id); const isFav = favList.includes(radio.id); const hasError = cardError && cardError.id === radio.id; const item = document.createElement("div"); item.className = `station-item ${isPlaying ? 'active' : ''}`; item.dataset.id = radio.id; const coverBg = radio.logo ? `url(${radio.logo}) center/cover no-repeat` : `linear-gradient(135deg, var(--primary), var(--primary-dark))`; const tag = radio.tag || "电台"; const errorHtml = hasError ? `
${cardError.message}
` : `
`; item.innerHTML = `
${radio.name}
${radio.city || '未知'} ${tag}
${isAllCategory ? ` ${isPlaying ? ` ` : ` `} ` : ` ${isPlaying ? ` ` : ` `} `}
${errorHtml} `; item.addEventListener('click', (e) => { if (e.target.closest('.play-btn')) return; if (e.target.closest('.del-btn')) return; if (e.target.closest('.star-btn')) return; if (hasError) { clearCardError(); playRadio(radio.id); return; } playRadio(radio.id); }); container.appendChild(item); }); container.querySelectorAll('.play-btn').forEach(btn => { btn.addEventListener('click', (e) => { e.stopPropagation(); if (cardError) clearCardError(); playRadio(btn.dataset.id); }); }); container.querySelectorAll('.star-btn').forEach(btn => { btn.addEventListener('click', function(e) { e.stopPropagation(); const id = this.dataset.id; const idx = favList.indexOf(id); if (idx > -1) { favList.splice(idx, 1); } else { favList.push(id); } Storage.save("ansonradio_fav", favList); renderStationList(); updateFavButtonState(); }); }); container.querySelectorAll('.del-btn').forEach(btn => { btn.addEventListener('click', function(e) { e.stopPropagation(); const id = this.dataset.id; const radio = radioDataSource.find(r => r.id === id); if (!radio) return; if (!confirm(`确定删除「${radio.name}」吗？`)) return; deleteRadioById(id); renderStationList(); }); }); updatePlayButtonState(); updateFavButtonState(); }
